function configureAnimationSet(name, fileList)
	if (animTable[name] ~= nil) then
		for _, connection in pairs(animTable[name].connections) do
			connection:disconnect()
		end
	end
	animTable[name] = {}
	animTable[name].count = 0
	animTable[name].totalWeight = 0	
	animTable[name].connections = {}

	-- check for config values
	local config = script:FindFirstChild(name)
	if (config ~= nil) then
		--		print("Loading anims " .. name)
		table.insert(animTable[name].connections, config.ChildAdded:connect(function(child) configureAnimationSet(name, fileList) end))
		table.insert(animTable[name].connections, config.ChildRemoved:connect(function(child) configureAnimationSet(name, fileList) end))
		local idx = 1
		for _, childPart in pairs(config:GetChildren()) do
			if (childPart:IsA("Animation")) then
				table.insert(animTable[name].connections, childPart.Changed:connect(function(property) configureAnimationSet(name, fileList) end))
				animTable[name][idx] = {}
				animTable[name][idx].anim = childPart
				local weightObject = childPart:FindFirstChild("Weight")
				if (weightObject == nil) then
					animTable[name][idx].weight = 1
				else
					animTable[name][idx].weight = weightObject.Value
				end
				animTable[name].count = animTable[name].count + 1
				animTable[name].totalWeight = animTable[name].totalWeight + animTable[name][idx].weight
				--			print(name .. " [" .. idx .. "] " .. animTable[name][idx].anim.AnimationId .. " (" .. animTable[name][idx].weight .. ")")
				idx = idx + 1
			end
		end
	end

	-- fallback to defaults
	if (animTable[name].count <= 0) then
		for idx, anim in pairs(fileList) do
			animTable[name][idx] = {}
			animTable[name][idx].anim = Instance.new("Animation")
			animTable[name][idx].anim.Name = name
			animTable[name][idx].anim.AnimationId = anim.id
			animTable[name][idx].weight = anim.weight
			animTable[name].count = animTable[name].count + 1
			animTable[name].totalWeight = animTable[name].totalWeight + anim.weight
			--			print(name .. " [" .. idx .. "] " .. anim.id .. " (" .. anim.weight .. ")")
		end
	end
end

function keyFrameReachedFunc(frameName)
	if (frameName == "End") then

		local repeatAnim = currentAnim
		-- return to idle if finishing an emote
		if (emoteNames[repeatAnim] ~= nil and emoteNames[repeatAnim] == false) then
			repeatAnim = "idle"
		end

		local animSpeed = currentAnimSpeed
		playAnimation(repeatAnim, 0.0, Humanoid)
		setAnimationSpeed(animSpeed)
	end
end

-- Preload animations
function playAnimation(animName, transitionTime, humanoid) 

	local roll = math.random(1, animTable[animName].totalWeight) 
	local origRoll = roll
	local idx = 1
	while (roll > animTable[animName][idx].weight) do
		roll = roll - animTable[animName][idx].weight
		idx = idx + 1
	end
	--		print(animName .. " " .. idx .. " [" .. origRoll .. "]")
	local anim = animTable[animName][idx].anim

	-- switch animation		
	if (anim ~= currentAnimInstance) then

		if (currentAnimTrack ~= nil) then
			currentAnimTrack:Stop(transitionTime)
			currentAnimTrack:Destroy()
		end

		currentAnimSpeed = 1.0

		-- load it to the humanoid; get AnimationTrack
		currentAnimTrack = humanoid:LoadAnimation(anim)
		currentAnimTrack.Priority = Enum.AnimationPriority.Core

		-- play the animation
		currentAnimTrack:Play(transitionTime)
		currentAnim = animName
		currentAnimInstance = anim

		-- set up keyframe name triggers
		if (currentAnimKeyframeHandler ~= nil) then
			currentAnimKeyframeHandler:disconnect()
		end
		currentAnimKeyframeHandler = currentAnimTrack.KeyframeReached:connect(keyFrameReachedFunc)

	end

end

local lastTick = 0

function move(time)
	local amplitude = 1
	local frequency = 1
	local deltaTime = time - lastTick
	lastTick = time

	local climbFudge = 0
	local setAngles = false

	if (jumpAnimTime > 0) then
		jumpAnimTime = jumpAnimTime - deltaTime
	end

	if (pose == "FreeFall" and jumpAnimTime <= 0) then
		playAnimation("fall", fallTransitionTime, Humanoid)
	elseif (pose == "Seated") then
		playAnimation("sit", 0.5, Humanoid)
		return
	elseif (pose == "Running") then
		playAnimation("walk", 0.1, Humanoid)
	elseif (pose == "Dead" or pose == "GettingUp" or pose == "FallingDown" or pose == "Seated" or pose == "PlatformStanding") then
		--		print("Wha " .. pose)
		stopAllAnimations()
		amplitude = 0.1
		frequency = 1
		setAngles = true
	end

	if (setAngles) then
		local desiredAngle = amplitude * math.sin(time * frequency)

		RightShoulder:SetDesiredAngle(desiredAngle + climbFudge)
		LeftShoulder:SetDesiredAngle(desiredAngle - climbFudge)
		RightHip:SetDesiredAngle(-desiredAngle)
		LeftHip:SetDesiredAngle(-desiredAngle)
	end

	-- Tool Animation handling
	local tool = getTool()
	if tool and tool:FindFirstChild("Handle") then

		local animStringValueObject = getToolAnim(tool)

		if animStringValueObject then
			toolAnim = animStringValueObject.Value
			-- message recieved, delete StringValue
			animStringValueObject.Parent = nil
			toolAnimTime = time + .3
		end

		if time > toolAnimTime then
			toolAnimTime = 0
			toolAnim = "None"
		end

		animateTool()		
	else
		stopToolAnimations()
		toolAnim = "None"
		toolAnimInstance = nil
		toolAnimTime = 0
	end
end

local function angleBetween(vectorA: Vector3, vectorB: Vector3)
	return math.acos(math.clamp(vectorA:Dot(vectorB), -1, 1))
end

local function isInFOV(target: BasePart, fov: number)
	local lookForward = humanoid.RootPart.CFrame.LookVector
	local lookToPoint = (target.CFrame.Position - humanoid.RootPart.Position).unit
	local angle = angleBetween(lookForward, lookToPoint)
	return math.abs(angle) <= math.rad(fov)
end

local function isGun(tool: Tool)
	return tool:GetAttribute("Arms") and not tool:GetAttribute("Melee")
end

local function isMelee(tool: Tool)
	return tool:GetAttribute("Arms") and tool:GetAttribute("Melee")
end

local function detectTarget(part: BasePart)
	return isInFOV(part, config.fov) or getDistance(part, humanoid.RootPart) <= config.targetCloseRangeNotice
end

local function getTargets(findRadius: number, ignoreObstacles: boolean, ignoreFOV: boolean, isAllies: boolean, includeDeadTarget: boolean)
	local targets = {}
	local hitted = {}
	local collideParts = workspace:GetPartBoundsInRadius(humanoid.RootPart.Position, findRadius, config.targetOverlapParams)
	for _,part in pairs(collideParts) do
		local target = isHumanoid(part)
		local params = RaycastParams.new()
		params.FilterDescendantsInstances = {character, target or part, workspace.Ignored}
		if target and target:IsDescendantOf(config.characters) and target ~= character and target:FindFirstChild("Humanoid") and (not target.Humanoid:GetAttribute("Team") or not isAllies and target.Humanoid:GetAttribute("Team") ~= humanoid:GetAttribute("Team") or isAllies and target.Humanoid:GetAttribute("Team") == humanoid:GetAttribute("Team")) and target.Humanoid.RootPart and (target.Humanoid.Health > 0 or includeDeadTarget) and not hitted[target] and part.Parent == target and not target:FindFirstChildWhichIsA("ForceField") and (ignoreObstacles or not isObstacle(humanoid.RootPart, part, params)) and (isInFOV(part, config.fov) or ignoreFOV or getDistance(part, humanoid.RootPart) <= config.targetCloseRangeNotice) then
			hitted[target] = true
			targets[#targets + 1] = target
		end
	end
	
	table.clear(hitted)
	
	return targets
end

local function findClosedTarget(findRadius: number, ignoreObstacles: boolean, ignoreFOV: boolean, isAllies: boolean, includeDeadTarget: boolean)
	if humanoid.RootPart then
		local target, dist = nil, math.huge
		
		local targets = getTargets(findRadius, ignoreObstacles, ignoreFOV, isAllies, includeDeadTarget)		
		for _,v in pairs(targets) do
			local targetDist = getDistance(v.Humanoid.RootPart, humanoid.RootPart)
			if targetDist < dist then
				target = v
				dist = targetDist
			end
		end
		
		table.clear(targets)
		
		return target, dist
	end
	
	return false
end

local function checkLadder(part: BasePart)
	local overlapParams = OverlapParams.new()
	overlapParams.FilterDescendantsInstances = {config.characters, workspace.Ignored}
	local collideParts = workspace:GetPartBoundsInRadius(part and part.Position or humanoid.RootPart.Position, 3, overlapParams)
	for _,v in pairs(collideParts) do
		if v:IsA("TrussPart") then
			return v
		end
	end
	
	return false
end

local function canJump(params: RaycastParams?)
	return not isObstacle(humanoid.RootPart, humanoid.RootPart.Position + Vector3.new(0, humanoid.JumpPower ^ 2 / (2 * workspace.Gravity), 0), params)
end

local function unequipTools()
	for _,v in pairs(character:GetChildren()) do
		if v:IsA("Tool") then
			v.Parent = backpack
		end
	end
end

local function equipTool(tool: Tool | string)
	unequipTools()
	if typeof(tool) == "Instance" then
		tool.Parent = character
	else
		local backpackTool = backpack:FindFirstChild(tool)
		if backpackTool then
			backpackTool.Parent = character
		end
	end
end

local function stopPaths(hash: string)
	if hash then
		activePaths[hash] = false
	else
		for hash,_ in pairs(activePaths) do
			activePaths[hash] = false
		end
	end
end

local function moveTo(point: Vector3, timeout: number)
	task.spawn(function()
		stopPaths()
		local hash = HttpService:GenerateGUID(false)
		activePaths[hash] = true

		local status = {
			reached = false,
			blocked = false,
			error = false,

			skipped = false,
		}

		local connections = {}
		table.insert(connections, path.Reached:Connect(function()
			status.reached = true
		end))

		table.insert(connections, path.Blocked:Connect(function(_, waypoint)
			status.blocked = true
			status.skipped = true
			
			humanoid.Jump = true
		end))

		table.insert(connections, path.Error:Connect(function(errorType)
			status.error = true
			status.skipped = true
			
			if errorType == "AgentStuck" and humanoid:GetState() ~= Enum.HumanoidStateType.Climbing then
				humanoid.Jump = true
			end
		end))
		
		if humanoid.Sit then
			humanoid.Jump = true
		end

		if not character:GetAttribute("meleeFinisher") then
			pcall(function() 
				local computed = path:Run(point)
			end)
		end

		local t = tick()
		while tick() - t < timeout and not status.reached and not status.skipped and activePaths[hash] do
			task.wait(config.rate)
		end
		
		_disconnectEvents(connections)
		table.clear(status)
		activePaths[hash] = nil
	end)
end

local function findCover(fromPart: BasePart, fromPartModel: Model)
	local coverSpots = {}
	for _,pos in pairs(config.findCoverGrid) do
		local origin = (humanoid.RootPart.CFrame * CFrame.new(pos)).p
		local params = RaycastParams.new()
		params.FilterDescendantsInstances = {fromPartModel or fromPart, character, workspace.Ignored, config.characters}
		local cast = workspace:Raycast(origin, fromPart.CFrame.LookVector * -config.findCoverDist, params)
		if config.debugMode then
			if cast then
				debug:debugLine(origin, cast.Position - origin, 0.5)
			else
				debug:debugLine(origin, fromPart.CFrame.LookVector * -config.findCoverDist, 0.5, nil, nil, Color3.fromRGB(255, 0, 0))
			end
		end
		
		local params = RaycastParams.new()
		params.FilterDescendantsInstances = {character, fromPartModel or fromPart, workspace.Ignored}
		local addPos = (humanoid.RootPart.CFrame * CFrame.new(Vector3.new(pos.x, 0, 0).unit * 2.5)).p - humanoid.RootPart.Position
		if cast and isObstacle(fromPart.Position, cast.Position + cast.Normal + addPos, params) then
			coverSpots[#coverSpots + 1] = {cast, addPos}
		end
	end
	
	local spot, pos, dist = nil, nil, math.huge
	for _,v in pairs(coverSpots) do
		local spotDist = getDistance(v[1].Position, humanoid.RootPart)
		if spotDist < dist then
			spot = v[1]
			pos = v[2]
			dist = spotDist
		end
	end
	
	return spot, pos
end

local function takeCover(jump: boolean, reload: boolean, rush: boolean)
	if not states.coverDamage and states.target then
		states.coverDamage = true
		local cover,pos = findCover(states.target.part, states.target.character)
		if cover then
			if rush then
				states.run = true
			end
			local params = RaycastParams.new()
			params.FilterDescendantsInstances = {character, states.target.character, workspace.Ignored}
			if jump and math.random(2) == 1 and canJump(params) then
				humanoid.Jump = true
			end
			local dist = cover.Position + cover.Normal + pos
			local timeout = getDistance(dist, humanoid.RootPart) + 0.5
			moveTo(dist, 1)
			local t = tick()
			task.wait(timeout / humanoid.WalkSpeed)
			local reloaded = false
			if reload and equippedTool and (not states.target or isObstacle(humanoid.RootPart, states.target.part, params)) then
				reloaded = true
				equippedTool.activateEvent:Fire(Enum.KeyCode.R, "began")
			elseif not reload and (not states.target or isObstacle(humanoid.RootPart, states.target.part, params)) then
				states.run = false
			end
			
			local strafeTick = 0
			local strafeSwapTick = 0
			local strafePos = math.random(1, 2)
			while tick() - t < 3 do
				local coverTarget,_ = findClosedTarget(config.findTargetRadius)
				if coverTarget then
					if not states.target or states.target.character ~= coverTarget then
						states.target = {part = coverTarget.Humanoid.RootPart, character = coverTarget}
						params.FilterDescendantsInstances = {character, states.target.character, workspace.Ignored}
					end
					if coverTarget.Humanoid:GetAttribute("Downed") then
						break
					end
					local params = RaycastParams.new()
					params.FilterDescendantsInstances = {character, coverTarget, workspace.Ignored}
					if getDistance(coverTarget.Humanoid.RootPart, humanoid.RootPart) > 5 and not isObstacle(humanoid.RootPart, coverTarget.Humanoid.RootPart, params) then
						local cover,pos = findCover(coverTarget.Humanoid.RootPart, coverTarget)
						if cover then
							local dist = cover.Position + cover.Normal + pos
							local timeout = getDistance(dist, humanoid.RootPart) + 0.5
							if config.debugMode then
								debug:point(dist, 1.5)
							end
							moveTo(dist, 1)
							task.wait(timeout / humanoid.WalkSpeed)
							if reload and equippedTool and not reloaded and (not coverTarget:IsDescendantOf(workspace) or not coverTarget.Humanoid.RootPart or isObstacle(humanoid.RootPart, coverTarget.Humanoid.RootPart, params)) then
								equippedTool.activateEvent:Fire(Enum.KeyCode.R, "began")
								states.run = false
							elseif not reload and (not coverTarget:IsDescendantOf(workspace) or not coverTarget.Humanoid.RootPart or isObstacle(humanoid.RootPart, coverTarget.Humanoid.RootPart, params)) then
								states.run = false
							end
							t = tick()
						else
							moveTo(clampDistanceBetweenVectors(coverTarget.Humanoid.RootPart.Position, humanoid.RootPart.Position, 20), 1)
						end
					else
						if canJump(params) and math.random(5) == 1 then
							humanoid.Jump = true
						end
						moveTo(clampDistanceBetweenVectors(coverTarget.Humanoid.RootPart.Position, humanoid.RootPart.Position, 20), 1)
					end
				else
					local strafeDir = {
						humanoid.RootPart.CFrame.RightVector * 5, 
						humanoid.RootPart.CFrame.RightVector * -5,
					}

					if tick() - strafeTick >= 0.5 and not character:GetAttribute("meleeFinisher") then
						strafeTick = tick()
						humanoid:MoveTo(humanoid.RootPart.Position + strafeDir[math.random(#strafeDir)] + strafeDir[strafePos])
						if tick() - strafeSwapTick >= 3 then
							strafeSwapTick = tick()
							strafePos = math.random(1, 2)
						end
					end
				end

				task.wait(config.rate)
			end
		else
			moveTo(clampDistanceBetweenVectors(states.target.part.Position, humanoid.RootPart.Position, 100), 1)
			if reload and equippedTool then
				equippedTool.activateEvent:Fire(Enum.KeyCode.R, "began")
			end
		end
		states.coverDamage = false
	elseif reload and equippedTool then
		equippedTool.activateEvent:Fire(Enum.KeyCode.R, "began")
	end
end

local function enemy()
	local target,_ = findClosedTarget(config.findTargetRadius, false, tick() - states.damageAlertTick < 0.5)
	if target then
		states.alertTick = tick()
		stopPaths()
		states.target = {part = target.Humanoid.RootPart, character = target}
		local params = RaycastParams.new()
		params.FilterDescendantsInstances = {character, states.target.character, workspace.Ignored}
		local takeCoverFast = getDistance(states.target.part, humanoid.RootPart) <= 15 and (not equippedTool or isGun(equippedTool))
		if math.random(2) == 1 or takeCoverFast or #getTargets(config.findTargetRadius * 1.5) >= config.fallBackEnemyCountOnFight and #getTargets(config.findTargetRadius, false, true, true) < config.stayAlliesCountOnFight then
			if not target.Humanoid:GetAttribute("Downed") then
				takeCover(false, false, takeCoverFast and true)
			end
		end
		local onSightTick = tick()
		
		local strafeTick = 0
		local strafeSwapTick = 0
		local strafePos = math.random(1, 2)
		
		local lastSeenPosition = states.target.part.Position
		local lostSightRadius = Vector3.new(math.random(-5, 5), 0, math.random(-5, 5))
		character:SetAttribute("aim", true)
		lastTargetAlertPos = states.target.part.Position
		local originRootPos = humanoid.RootPart.Position
		if equippedTool and isMelee(equippedTool) then
			states.run = true
		end
		while target:IsDescendantOf(workspace) and target.Humanoid.Health > 0 and target.Humanoid.RootPart and getDistance(target.Humanoid.RootPart, humanoid.RootPart) < config.forgetTargetOnFarDistance do
			local onSight = not isObstacle(humanoid.RootPart, states.target.part, params)
			if onSight and detectTarget(states.target.part) or checkLadder() then
				onSightTick = tick()
				lastSeenPosition = states.target.part.Position
				lostSightRadius = Vector3.new(math.random(-5, 5), 0, math.random(-5, 5))
			elseif tick() - onSightTick >= config.forgetTargetOnSightLost then
				lastTargetPos = lastSeenPosition
				lastTargetAlertPos = lastSeenPosition
				break
			--[[elseif not onSight and target.Humanoid.Health > 50 and equippedTool and isGun(equippedTool) then
				takeCover(false, false, true)]]
			end
			
			if math.random(5) == 1 and not states.target.character.Humanoid:GetAttribute("Downed") and #getTargets(config.findTargetRadius) >= config.fallBackEnemyCountOnFight and #getTargets(config.findTargetRadius, false, true, true) < config.stayAlliesCountOnFight or humanoid:GetAttribute("Downed") then
				takeCover(false, false, true)
			end
			
			if equippedTool and isGun(equippedTool) and not states.target.character.Humanoid:GetAttribute("Downed") and ((equippedTool._data.magCurrent.Value / equippedTool._data.magSize.Value) * 1 <= 0.3 and math.random(3) == 1 or equippedTool._data.magCurrent.Value <= 0) and equippedTool._data.ammo.Value > 0 then
				takeCover(false, true, true)
			end
			
			local otherCloserTarget,_ = findClosedTarget(config.findTargetRadius)
			if otherCloserTarget and otherCloserTarget ~= target then
				target = otherCloserTarget
				states.target = {part = target.Humanoid.RootPart, character = target}
				params.FilterDescendantsInstances = {character, states.target.character, workspace.Ignored}
				originRootPos = humanoid.RootPart.Position
			end
			
			local ladderPart = checkLadder()
			local targetIsClimbing = checkLadder(states.target.part) == ladderPart and ladderPart ~= false
			if targetIsClimbing then
				moveTo(ladderPart.Position + Vector3.new(0, ladderPart.Size.y / 2, 0), 1)
			else
				if equippedTool and isGun(equippedTool) then
					local strafeDir = {
						humanoid.RootPart.CFrame.RightVector * 5, 
						humanoid.RootPart.CFrame.RightVector * -5,
					}
					
					if getDistance(originRootPos, lastSeenPosition) < config.gunMinCloseFromOriginDist then
						if onSight then
							moveTo(clampDistanceBetweenVectors(lastSeenPosition + strafeDir[math.random(#strafeDir)], originRootPos, config.gunMinCloseFromOriginDist), 1)
						else
							moveTo(lastSeenPosition + lostSightRadius + strafeDir[math.random(#strafeDir)], 1)
						end
					elseif tick() - strafeTick >= 0.5 and not character:GetAttribute("meleeFinisher") then
						strafeTick = tick()
						humanoid:MoveTo(humanoid.RootPart.Position + strafeDir[math.random(#strafeDir)] + strafeDir[strafePos])
						if tick() - strafeSwapTick >= 3 then
							strafeSwapTick = tick()
							strafePos = math.random(1, 2)
						end
					end
				elseif equippedTool and isMelee(equippedTool) then
					moveTo(onSight and clampDistanceBetweenVectors(lastSeenPosition, humanoid.RootPart.Position, config.meleeMinCloseDist) or lastSeenPosition + lostSightRadius, 1)
				else
					moveTo(onSight and lastSeenPosition or lastSeenPosition + lostSightRadius, 1)
				end
			end
			while states.coverDamage do
				if target:IsDescendantOf(workspace) and target.Humanoid.Health > 0 and target.Humanoid.RootPart and not isObstacle(humanoid.RootPart, states.target.part, params) then
					onSightTick = tick()
				end
				task.wait(config.rate * 2)
			end
			if states.target and target ~= states.target.character then
				target = states.target.character
			end
			
			local waitTime = checkLadder() and humanoid:GetState() == Enum.HumanoidStateType.Climbing and math.max(getDistance(lastSeenPosition, humanoid.RootPart) / 40, 3) or ladderPart and 1 or config.rate
			if waitTime <= config.rate then
				task.wait(config.rate)
			else
				local t = tick()
				while tick() - t < waitTime do
					if getDistance(lastSeenPosition, humanoid.RootPart) <= 1 then
						break
					end
					if states.target.part.Position.y < humanoid.RootPart.Position.y - 5 and not isObstacle(humanoid.RootPart, states.target.part, params) then
						humanoid.Jump = true
						task.wait()
						break
					end
					task.wait(config.rate)
				end
			end
		end
		
		character:SetAttribute("aim", false)
		table.clear(states.target)
		states.target = false
		states.run = false
		return true
	else
		if typeof(states.target) == "table" then
			table.clear(states.target)
		end
		states.target = false
	end
	
	return false
end

Path.__index = function(table, index)

	end

--[[ PRIVATE FUNCTIONS ]]--
local function declareError(self, errorType)
	self._lastError = errorType
	self._events.Error:Fire(errorType)
end

--Create visual waypoints
local function createVisualWaypoints(waypoints)
	local visualWaypoints = {}
	for _, waypoint in ipairs(waypoints) do
		local visualWaypointClone = visualWaypoint:Clone()
		visualWaypointClone.Position = waypoint.Position
		visualWaypointClone.Parent = workspace
		visualWaypointClone.Color =
			(waypoint == waypoints[#waypoints] and Color3.fromRGB(0, 255, 0))
			or (waypoint.Action == Enum.PathWaypointAction.Jump and Color3.fromRGB(255, 0, 0))
			or Color3.fromRGB(255, 139, 0)
		table.insert(visualWaypoints, visualWaypointClone)
	end
	return visualWaypoints
end

--Destroy visual waypoints
local function destroyVisualWaypoints(waypoints)
	if waypoints then
		for _, waypoint in ipairs(waypoints) do
			waypoint:Destroy()
		end
	end
	return
end

--Get initial waypoint for non-humanoid
local function getNonHumanoidWaypoint(self)
	--Account for multiple waypoints that are sometimes in the same place
	for i = 2, #self._waypoints do
		if (self._waypoints[i].Position - self._waypoints[i - 1].Position).Magnitude > 0.1 then
			return i
		end
	end
	return 2
end

--Make NPC jump
local function setJumpState(self)
	pcall(function()
		if self._humanoid:GetState() ~= Enum.HumanoidStateType.Jumping and self._humanoid:GetState() ~= Enum.HumanoidStateType.Freefall then
			self._humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
		end
	end)
end

--Primary move function
local function move(self)
	if self._waypoints[self._currentWaypoint].Action == Enum.PathWaypointAction.Jump then
		setJumpState(self)
	end
	self._humanoid:MoveTo(self._waypoints[self._currentWaypoint].Position)
end

--Disconnect MoveToFinished connection when pathfinding ends
local function disconnectMoveConnection(self)
	self._moveConnection:Disconnect()
	self._moveConnection = nil
end

--Fire the WaypointReached event
local function invokeWaypointReached(self)
	local lastWaypoint = self._waypoints[self._currentWaypoint - 1]
	local nextWaypoint = self._waypoints[self._currentWaypoint]
	self._events.WaypointReached:Fire(self._agent, lastWaypoint, nextWaypoint)
end

local function moveToFinished(self, reached)
	
	--Stop execution if Path is destroyed
	if not getmetatable(self) then return end

	--Handle case for non-humanoids
	if not self._humanoid then
		if reached and self._currentWaypoint + 1 <= #self._waypoints then
			invokeWaypointReached(self)
			self._currentWaypoint += 1
		elseif reached then
			self._visualWaypoints = destroyVisualWaypoints(self._visualWaypoints)
			self._target = nil
			self._events.Reached:Fire(self._agent, self._waypoints[self._currentWaypoint])
		else
			self._visualWaypoints = destroyVisualWaypoints(self._visualWaypoints)
			self._target = nil
			declareError(self, self.ErrorType.TargetUnreachable)
		end
		return
	end

	if reached and self._currentWaypoint + 1 <= #self._waypoints  then --Waypoint reached
		if self._currentWaypoint + 1 < #self._waypoints then
			invokeWaypointReached(self)
		end
		self._currentWaypoint += 1
		move(self)
	elseif reached then --Target reached, pathfinding ends
		disconnectMoveConnection(self)
		self._status = Path.StatusType.Idle
		self._visualWaypoints = destroyVisualWaypoints(self._visualWaypoints)
		self._events.Reached:Fire(self._agent, self._waypoints[self._currentWaypoint])
	else --Target unreachable
		disconnectMoveConnection(self)
		self._status = Path.StatusType.Idle
		self._visualWaypoints = destroyVisualWaypoints(self._visualWaypoints)
		declareError(self, self.ErrorType.TargetUnreachable)
	end
end

--Refer to Settings.COMPARISON_CHECKS
local function comparePosition(self)
	if self._currentWaypoint == #self._waypoints then return end
	self._position._count = ((self._agent.PrimaryPart.Position - self._position._last).Magnitude <= 0.07 and (self._position._count + 1)) or 0
	self._position._last = self._agent.PrimaryPart.Position
	if self._position._count >= self._settings.COMPARISON_CHECKS then
		if self._settings.JUMP_WHEN_STUCK then
			setJumpState(self)
		end
		declareError(self, self.ErrorType.AgentStuck)
	end
end

--[[ STATIC METHODS ]]--
function Path.GetNearestCharacter(fromPosition)
	local character, dist = nil, math.huge
	for _, player in ipairs(Players:GetPlayers()) do
		if player.Character and (player.Character.PrimaryPart.Position - fromPosition).Magnitude < dist then
			character, dist = player.Character, (player.Character.PrimaryPart.Position - fromPosition).Magnitude
		end
	end
	return character
end

--[[ CONSTRUCTOR ]]--
function Path.new(agent, agentParameters, override)
	if not (agent and agent:IsA("Model") and agent.PrimaryPart) then
		output(error, "Pathfinding agent must be a valid Model Instance with a set PrimaryPart.")
	end

	local self = setmetatable({
		_settings = override or DEFAULT_SETTINGS;
		_events = {
			Reached = Instance.new("BindableEvent");
			WaypointReached = Instance.new("BindableEvent");
			Blocked = Instance.new("BindableEvent");
			Error = Instance.new("BindableEvent");
			Stopped = Instance.new("BindableEvent");
		};
		_agent = agent;
		_humanoid = agent:FindFirstChildOfClass("Humanoid");
		_path = PathfindingService:CreatePath(agentParameters);
		_status = "Idle";
		_t = 0;
		_position = {
			_last = Vector3.new();
			_count = 0;
		};
	}, Path)

	--Configure settings
	for setting, value in pairs(DEFAULT_SETTINGS) do
		self._settings[setting] = self._settings[setting] == nil and value or self._settings[setting]
	end

	--Path blocked connection
	self._path.Blocked:Connect(function(...)
		if (self._currentWaypoint <= ... and self._currentWaypoint + 1 >= ...) and self._humanoid then
			setJumpState(self)
			self._events.Blocked:Fire(self._agent, self._waypoints[...])
		end
	end)

	return self
end


--[[ NON-STATIC METHODS ]]--
function Path:Destroy()
	for _, event in ipairs(self._events) do
		event:Destroy()
	end
	self._events = nil
	if rawget(self, "_visualWaypoints") then
		self._visualWaypoints = destroyVisualWaypoints(self._visualWaypoints)
	end
	self._path:Destroy()
	setmetatable(self, nil)
	for k, _ in pairs(self) do
		self[k] = nil
	end
end

function Path:Stop()
	if not self._humanoid then
		output(error, "Attempt to call Path:Stop() on a non-humanoid.")
		return
	end
	if self._status == Path.StatusType.Idle then
		output(function(m)
			warn(debug.traceback(m))
		end, "Attempt to run Path:Stop() in idle state")
		return
	end
	disconnectMoveConnection(self)
	self._status = Path.StatusType.Idle
	self._visualWaypoints = destroyVisualWaypoints(self._visualWaypoints)
	self._events.Stopped:Fire(self._model)
end

function Path:Run(target)

	--Non-humanoid handle case
	if not target and not self._humanoid and self._target then
		moveToFinished(self, true)
		return
	end

	--Parameter check
	if not (target and (typeof(target) == "Vector3" or target:IsA("BasePart"))) then
		output(error, "Pathfinding target must be a valid Vector3 or BasePart.")
	end

	--Refer to Settings.TIME_VARIANCE
	if os.clock() - self._t <= self._settings.TIME_VARIANCE and self._humanoid then
		task.wait(os.clock() - self._t)
		declareError(self, self.ErrorType.LimitReached)
		return false
	elseif self._humanoid then
		self._t = os.clock()
	end

	--Compute path
	local pathComputed, _ = pcall(function()
		self._path:ComputeAsync(self._agent.PrimaryPart.Position, (typeof(target) == "Vector3" and target) or target.Position)
	end)

	--Make sure path computation is successful
	if not pathComputed
		or self._path.Status == Enum.PathStatus.NoPath
		or #self._path:GetWaypoints() < 2
		or (self._humanoid and self._humanoid:GetState() == Enum.HumanoidStateType.Freefall) then
		self._visualWaypoints = destroyVisualWaypoints(self._visualWaypoints)
		task.wait()
		declareError(self, self.ErrorType.ComputationError)
		return false
	end

	--Set status to active; pathfinding starts
	self._status = (self._humanoid and Path.StatusType.Active) or Path.StatusType.Idle
	self._target = target

	--Set network owner to server to prevent "hops"
	pcall(function()
		self._agent.PrimaryPart:SetNetworkOwner(nil)
	end)

	--Initialize waypoints
	self._waypoints = self._path:GetWaypoints()
	self._currentWaypoint = 2

	--Refer to Settings.COMPARISON_CHECKS
	if self._humanoid then
		comparePosition(self)
	end

	--Visualize waypoints
	destroyVisualWaypoints(self._visualWaypoints)
	self._visualWaypoints = (self.Visualize and createVisualWaypoints(self._waypoints))

	--Create a new move connection if it doesn't exist already
	self._moveConnection = self._humanoid and (self._moveConnection or self._humanoid.MoveToFinished:Connect(function(...)
		moveToFinished(self, ...)
	end))

	--Begin pathfinding
	if self._humanoid then
		self._humanoid:MoveTo(self._waypoints[self._currentWaypoint].Position)
	elseif #self._waypoints == 2 then
		self._target = nil
		self._visualWaypoints = destroyVisualWaypoints(self._visualWaypoints)
		self._events.Reached:Fire(self._agent, self._waypoints[2])
	else
		self._currentWaypoint = getNonHumanoidWaypoint(self)
		moveToFinished(self, true)
	end
	return true
end

function isHumanoid(part: BasePart)
	if part and part.Parent and part.Parent:FindFirstChild("Humanoid") then
		return part.Parent
	elseif part and part.Parent and part.Parent ~= workspace then
		return isHumanoid(part.Parent)
	end

	return false
end

local function isObstacle(p1, p2, params)
	local cast = workspace:Raycast((typeof(p1) == "Instance" and p1.Position or p1), (typeof(p2) == "Instance" and p2.Position or p2) - (typeof(p1) == "Instance" and p1.Position or p1), params)
	if config.debugMode then
		if cast then
			debug:debugLine((typeof(p1) == "Instance" and p1.Position or p1), cast.Position - (typeof(p1) == "Instance" and p1.Position or p1), 1.5, Color3.fromRGB(0, 255, 0), Color3.fromRGB(0, 255, 0))
		else
			debug:debugLine((typeof(p1) == "Instance" and p1.Position or p1), (typeof(p2) == "Instance" and p2.Position or p2) - (typeof(p1) == "Instance" and p1.Position or p1), 1.5, nil, nil, Color3.fromRGB(255, 0, 0))
		end
	end
	
	return cast
end

local function getRandomChild(parent)
	return parent:GetChildren()[math.random(#parent:GetChildren())]
end

local function getDistance(p1, p2)
	return ((typeof(p1) == "Instance" and p1.Position or p1) - (typeof(p2) == "Instance" and p2.Position or p2)).magnitude
end

local function clampDistanceBetweenVectors(V1, V2, Max) -- V1 is origin.
	return V1 + ((V2 - V1).Unit * math.min((V2 - V1).Magnitude,  Max))
end

local function angleBetween(vectorA: Vector3, vectorB: Vector3)
	return math.acos(math.clamp(vectorA:Dot(vectorB), -1, 1))
end

local function isInFOV(target: BasePart, fov: number)
	local lookForward = humanoid.RootPart.CFrame.LookVector
	local lookToPoint = (target.CFrame.Position - humanoid.RootPart.Position).unit
	local angle = angleBetween(lookForward, lookToPoint)
	return math.abs(angle) <= math.rad(fov)
end

local function isGun(tool: Tool)
	return tool:GetAttribute("Arms") and not tool:GetAttribute("Melee")
end

local function isMelee(tool: Tool)
	return tool:GetAttribute("Arms") and tool:GetAttribute("Melee")
end

local function detectTarget(part: BasePart)
	return isInFOV(part, config.fov) or getDistance(part, humanoid.RootPart) <= config.targetCloseRangeNotice
end

local function getTargets(findRadius: number, ignoreObstacles: boolean, ignoreFOV: boolean, isAllies: boolean, includeDeadTarget: boolean)
	local targets = {}
	local hitted = {}
	local collideParts = workspace:GetPartBoundsInRadius(humanoid.RootPart.Position, findRadius, config.targetOverlapParams)
	for _,part in pairs(collideParts) do
		local target = isHumanoid(part)
		local params = RaycastParams.new()
		params.FilterDescendantsInstances = {character, target or part, workspace.Ignored}
		if target and target:IsDescendantOf(config.characters) and target ~= character and target:FindFirstChild("Humanoid") and (not target.Humanoid:GetAttribute("Team") or not isAllies and target.Humanoid:GetAttribute("Team") ~= humanoid:GetAttribute("Team") or isAllies and target.Humanoid:GetAttribute("Team") == humanoid:GetAttribute("Team")) and target.Humanoid.RootPart and (target.Humanoid.Health > 0 or includeDeadTarget) and not hitted[target] and part.Parent == target and not target:FindFirstChildWhichIsA("ForceField") and (ignoreObstacles or not isObstacle(humanoid.RootPart, part, params)) and (isInFOV(part, config.fov) or ignoreFOV or getDistance(part, humanoid.RootPart) <= config.targetCloseRangeNotice) then
			hitted[target] = true
			targets[#targets + 1] = target
		end
	end
	
	table.clear(hitted)
	
	return targets
end

local function findClosedTarget(findRadius: number, ignoreObstacles: boolean, ignoreFOV: boolean, isAllies: boolean, includeDeadTarget: boolean)
	if humanoid.RootPart then
		local target, dist = nil, math.huge
		
		local targets = getTargets(findRadius, ignoreObstacles, ignoreFOV, isAllies, includeDeadTarget)		
		for _,v in pairs(targets) do
			local targetDist = getDistance(v.Humanoid.RootPart, humanoid.RootPart)
			if targetDist < dist then
				target = v
				dist = targetDist
			end
		end
		
		table.clear(targets)
		
		return target, dist
	end
	
	return false
end

local function checkLadder(part: BasePart)
	local overlapParams = OverlapParams.new()
	overlapParams.FilterDescendantsInstances = {config.characters, workspace.Ignored}
	local collideParts = workspace:GetPartBoundsInRadius(part and part.Position or humanoid.RootPart.Position, 3, overlapParams)
	for _,v in pairs(collideParts) do
		if v:IsA("TrussPart") then
			return v
		end
	end
	
	return false
end

local function canJump(params: RaycastParams?)
	return not isObstacle(humanoid.RootPart, humanoid.RootPart.Position + Vector3.new(0, humanoid.JumpPower ^ 2 / (2 * workspace.Gravity), 0), params)
end

local function stopPaths(hash: string)
	if hash then
		activePaths[hash] = false
	else
		for hash,_ in pairs(activePaths) do
			activePaths[hash] = false
		end
	end
end

local function moveTo(point: Vector3, timeout: number)
	task.spawn(function()
		stopPaths()
		local hash = HttpService:GenerateGUID(false)
		activePaths[hash] = true

		local status = {
			reached = false,
			blocked = false,
			error = false,

			skipped = false,
		}

		local connections = {}
		table.insert(connections, path.Reached:Connect(function()
			status.reached = true
		end))

		table.insert(connections, path.Blocked:Connect(function(_, waypoint)
			status.blocked = true
			status.skipped = true
			
			humanoid.Jump = true
		end))

		table.insert(connections, path.Error:Connect(function(errorType)
			status.error = true
			status.skipped = true
			
			if errorType == "AgentStuck" and humanoid:GetState() ~= Enum.HumanoidStateType.Climbing then
				humanoid.Jump = true
			end
		end))
		
		if humanoid.Sit then
			humanoid.Jump = true
		end

		if not character:GetAttribute("meleeFinisher") then
			pcall(function() 
				local computed = path:Run(point)
			end)
		end

		local t = tick()
		while tick() - t < timeout and not status.reached and not status.skipped and activePaths[hash] do
			task.wait(config.rate)
		end
		
		_disconnectEvents(connections)
		table.clear(status)
		activePaths[hash] = nil
	end)
end

local function findCover(fromPart: BasePart, fromPartModel: Model)
	local coverSpots = {}
	for _,pos in pairs(config.findCoverGrid) do
		local origin = (humanoid.RootPart.CFrame * CFrame.new(pos)).p
		local params = RaycastParams.new()
		params.FilterDescendantsInstances = {fromPartModel or fromPart, character, workspace.Ignored, config.characters}
		local cast = workspace:Raycast(origin, fromPart.CFrame.LookVector * -config.findCoverDist, params)
		if config.debugMode then
			if cast then
				debug:debugLine(origin, cast.Position - origin, 0.5)
			else
				debug:debugLine(origin, fromPart.CFrame.LookVector * -config.findCoverDist, 0.5, nil, nil, Color3.fromRGB(255, 0, 0))
			end
		end
		
		local params = RaycastParams.new()
		params.FilterDescendantsInstances = {character, fromPartModel or fromPart, workspace.Ignored}
		local addPos = (humanoid.RootPart.CFrame * CFrame.new(Vector3.new(pos.x, 0, 0).unit * 2.5)).p - humanoid.RootPart.Position
		if cast and isObstacle(fromPart.Position, cast.Position + cast.Normal + addPos, params) then
			coverSpots[#coverSpots + 1] = {cast, addPos}
		end
	end
	
	local spot, pos, dist = nil, nil, math.huge
	for _,v in pairs(coverSpots) do
		local spotDist = getDistance(v[1].Position, humanoid.RootPart)
		if spotDist < dist then
			spot = v[1]
			pos = v[2]
			dist = spotDist
		end
	end
	
	return spot, pos
end

local function takeCover(jump: boolean, reload: boolean, rush: boolean)
	if not states.coverDamage and states.target then
		states.coverDamage = true
		local cover,pos = findCover(states.target.part, states.target.character)
		if cover then
			if rush then
				states.run = true
			end
			local params = RaycastParams.new()
			params.FilterDescendantsInstances = {character, states.target.character, workspace.Ignored}
			if jump and math.random(2) == 1 and canJump(params) then
				humanoid.Jump = true
			end
			local dist = cover.Position + cover.Normal + pos
			local timeout = getDistance(dist, humanoid.RootPart) + 0.5
			moveTo(dist, 1)
			local t = tick()
			task.wait(timeout / humanoid.WalkSpeed)
			local reloaded = false
			if reload and equippedTool and (not states.target or isObstacle(humanoid.RootPart, states.target.part, params)) then
				reloaded = true
				equippedTool.activateEvent:Fire(Enum.KeyCode.R, "began")
			elseif not reload and (not states.target or isObstacle(humanoid.RootPart, states.target.part, params)) then
				states.run = false
			end
			
			local strafeTick = 0
			local strafeSwapTick = 0
			local strafePos = math.random(1, 2)
			while tick() - t < 3 do
				local coverTarget,_ = findClosedTarget(config.findTargetRadius)
				if coverTarget then
					if not states.target or states.target.character ~= coverTarget then
						states.target = {part = coverTarget.Humanoid.RootPart, character = coverTarget}
						params.FilterDescendantsInstances = {character, states.target.character, workspace.Ignored}
					end
					if coverTarget.Humanoid:GetAttribute("Downed") then
						break
					end
					local params = RaycastParams.new()
					params.FilterDescendantsInstances = {character, coverTarget, workspace.Ignored}
					if getDistance(coverTarget.Humanoid.RootPart, humanoid.RootPart) > 5 and not isObstacle(humanoid.RootPart, coverTarget.Humanoid.RootPart, params) then
						local cover,pos = findCover(coverTarget.Humanoid.RootPart, coverTarget)
						if cover then
							local dist = cover.Position + cover.Normal + pos
							local timeout = getDistance(dist, humanoid.RootPart) + 0.5
							if config.debugMode then
								debug:point(dist, 1.5)
							end
							moveTo(dist, 1)
							task.wait(timeout / humanoid.WalkSpeed)
							if reload and equippedTool and not reloaded and (not coverTarget:IsDescendantOf(workspace) or not coverTarget.Humanoid.RootPart or isObstacle(humanoid.RootPart, coverTarget.Humanoid.RootPart, params)) then
								equippedTool.activateEvent:Fire(Enum.KeyCode.R, "began")
								states.run = false
							elseif not reload and (not coverTarget:IsDescendantOf(workspace) or not coverTarget.Humanoid.RootPart or isObstacle(humanoid.RootPart, coverTarget.Humanoid.RootPart, params)) then
								states.run = false
							end
							t = tick()
						else
							moveTo(clampDistanceBetweenVectors(coverTarget.Humanoid.RootPart.Position, humanoid.RootPart.Position, 20), 1)
						end
					else
						if canJump(params) and math.random(5) == 1 then
							humanoid.Jump = true
						end
						moveTo(clampDistanceBetweenVectors(coverTarget.Humanoid.RootPart.Position, humanoid.RootPart.Position, 20), 1)
					end
				else
					local strafeDir = {
						humanoid.RootPart.CFrame.RightVector * 5, 
						humanoid.RootPart.CFrame.RightVector * -5,
					}

					if tick() - strafeTick >= 0.5 and not character:GetAttribute("meleeFinisher") then
						strafeTick = tick()
						humanoid:MoveTo(humanoid.RootPart.Position + strafeDir[math.random(#strafeDir)] + strafeDir[strafePos])
						if tick() - strafeSwapTick >= 3 then
							strafeSwapTick = tick()
							strafePos = math.random(1, 2)
						end
					end
				end

				task.wait(config.rate)
			end
		else
			moveTo(clampDistanceBetweenVectors(states.target.part.Position, humanoid.RootPart.Position, 100), 1)
			if reload and equippedTool then
				equippedTool.activateEvent:Fire(Enum.KeyCode.R, "began")
			end
		end
		states.coverDamage = false
	elseif reload and equippedTool then
		equippedTool.activateEvent:Fire(Enum.KeyCode.R, "began")
	end
end

local function enemy()
	local target,_ = findClosedTarget(config.findTargetRadius, false, tick() - states.damageAlertTick < 0.5)
	if target then
		states.alertTick = tick()
		stopPaths()
		states.target = {part = target.Humanoid.RootPart, character = target}
		local params = RaycastParams.new()
		params.FilterDescendantsInstances = {character, states.target.character, workspace.Ignored}
		local takeCoverFast = getDistance(states.target.part, humanoid.RootPart) <= 15 and (not equippedTool or isGun(equippedTool))
		if math.random(2) == 1 or takeCoverFast or #getTargets(config.findTargetRadius * 1.5) >= config.fallBackEnemyCountOnFight and #getTargets(config.findTargetRadius, false, true, true) < config.stayAlliesCountOnFight then
			if not target.Humanoid:GetAttribute("Downed") then
				takeCover(false, false, takeCoverFast and true)
			end
		end
		local onSightTick = tick()
		
		local strafeTick = 0
		local strafeSwapTick = 0
		local strafePos = math.random(1, 2)
		
		local lastSeenPosition = states.target.part.Position
		local lostSightRadius = Vector3.new(math.random(-5, 5), 0, math.random(-5, 5))
		character:SetAttribute("aim", true)
		lastTargetAlertPos = states.target.part.Position
		local originRootPos = humanoid.RootPart.Position
		if equippedTool and isMelee(equippedTool) then
			states.run = true
		end
		while target:IsDescendantOf(workspace) and target.Humanoid.Health > 0 and target.Humanoid.RootPart and getDistance(target.Humanoid.RootPart, humanoid.RootPart) < config.forgetTargetOnFarDistance do
			local onSight = not isObstacle(humanoid.RootPart, states.target.part, params)
			if onSight and detectTarget(states.target.part) or checkLadder() then
				onSightTick = tick()
				lastSeenPosition = states.target.part.Position
				lostSightRadius = Vector3.new(math.random(-5, 5), 0, math.random(-5, 5))
			elseif tick() - onSightTick >= config.forgetTargetOnSightLost then
				lastTargetPos = lastSeenPosition
				lastTargetAlertPos = lastSeenPosition
				break
			--[[elseif not onSight and target.Humanoid.Health > 50 and equippedTool and isGun(equippedTool) then
				takeCover(false, false, true)]]
			end
			
			if math.random(5) == 1 and not states.target.character.Humanoid:GetAttribute("Downed") and #getTargets(config.findTargetRadius) >= config.fallBackEnemyCountOnFight and #getTargets(config.findTargetRadius, false, true, true) < config.stayAlliesCountOnFight or humanoid:GetAttribute("Downed") then
				takeCover(false, false, true)
			end
			
			if equippedTool and isGun(equippedTool) and not states.target.character.Humanoid:GetAttribute("Downed") and ((equippedTool._data.magCurrent.Value / equippedTool._data.magSize.Value) * 1 <= 0.3 and math.random(3) == 1 or equippedTool._data.magCurrent.Value <= 0) and equippedTool._data.ammo.Value > 0 then
				takeCover(false, true, true)
			end
			
			local otherCloserTarget,_ = findClosedTarget(config.findTargetRadius)
			if otherCloserTarget and otherCloserTarget ~= target then
				target = otherCloserTarget
				states.target = {part = target.Humanoid.RootPart, character = target}
				params.FilterDescendantsInstances = {character, states.target.character, workspace.Ignored}
				originRootPos = humanoid.RootPart.Position
			end
			
			local ladderPart = checkLadder()
			local targetIsClimbing = checkLadder(states.target.part) == ladderPart and ladderPart ~= false
			if targetIsClimbing then
				moveTo(ladderPart.Position + Vector3.new(0, ladderPart.Size.y / 2, 0), 1)
			else
				if equippedTool and isGun(equippedTool) then
					local strafeDir = {
						humanoid.RootPart.CFrame.RightVector * 5, 
						humanoid.RootPart.CFrame.RightVector * -5,
					}
					
					if getDistance(originRootPos, lastSeenPosition) < config.gunMinCloseFromOriginDist then
						if onSight then
							moveTo(clampDistanceBetweenVectors(lastSeenPosition + strafeDir[math.random(#strafeDir)], originRootPos, config.gunMinCloseFromOriginDist), 1)
						else
							moveTo(lastSeenPosition + lostSightRadius + strafeDir[math.random(#strafeDir)], 1)
						end
					elseif tick() - strafeTick >= 0.5 and not character:GetAttribute("meleeFinisher") then
						strafeTick = tick()
						humanoid:MoveTo(humanoid.RootPart.Position + strafeDir[math.random(#strafeDir)] + strafeDir[strafePos])
						if tick() - strafeSwapTick >= 3 then
							strafeSwapTick = tick()
							strafePos = math.random(1, 2)
						end
					end
				elseif equippedTool and isMelee(equippedTool) then
					moveTo(onSight and clampDistanceBetweenVectors(lastSeenPosition, humanoid.RootPart.Position, config.meleeMinCloseDist) or lastSeenPosition + lostSightRadius, 1)
				else
					moveTo(onSight and lastSeenPosition or lastSeenPosition + lostSightRadius, 1)
				end
			end
			while states.coverDamage do
				if target:IsDescendantOf(workspace) and target.Humanoid.Health > 0 and target.Humanoid.RootPart and not isObstacle(humanoid.RootPart, states.target.part, params) then
					onSightTick = tick()
				end
				task.wait(config.rate * 2)
			end
			if states.target and target ~= states.target.character then
				target = states.target.character
			end
			
			local waitTime = checkLadder() and humanoid:GetState() == Enum.HumanoidStateType.Climbing and math.max(getDistance(lastSeenPosition, humanoid.RootPart) / 40, 3) or ladderPart and 1 or config.rate
			if waitTime <= config.rate then
				task.wait(config.rate)
			else
				local t = tick()
				while tick() - t < waitTime do
					if getDistance(lastSeenPosition, humanoid.RootPart) <= 1 then
						break
					end
					if states.target.part.Position.y < humanoid.RootPart.Position.y - 5 and not isObstacle(humanoid.RootPart, states.target.part, params) then
						humanoid.Jump = true
						task.wait()
						break
					end
					task.wait(config.rate)
				end
			end
		end
		
		character:SetAttribute("aim", false)
		table.clear(states.target)
		states.target = false
		states.run = false
		return true
	else
		if typeof(states.target) == "table" then
			table.clear(states.target)
		end
		states.target = false
	end
	
	return false
end

local Path = {
	StatusType = {
		Idle = "Idle";
		Active = "Active";
	};
	ErrorType = {
		LimitReached = "LimitReached";
		TargetUnreachable = "TargetUnreachable";
		ComputationError = "ComputationError";

	};

Path.__index = function(table, index)
	if index == "Stopped" and not table._humanoid then
		output(error, "Attempt to use Path.Stopped on a non-humanoid.")
	end
	return (table._events[index] and table._events[index].Event)
		or (index == "LastError" and table._lastError)
		or (index == "Status" and table._status)
		or Path[index]
end

--Used to visualize waypoints
local visualWaypoint = Instance.new("Part")
visualWaypoint.Size = Vector3.new(0.3, 0.3, 0.3)
visualWaypoint.Anchored = true
visualWaypoint.CanCollide = false
visualWaypoint.Material = Enum.Material.Neon

--[[ PRIVATE FUNCTIONS ]]--
local function declareError(self, errorType)
	self._lastError = errorType
	self._events.Error:Fire(errorType)
end

--Create visual waypoints
local function createVisualWaypoints(waypoints)
	local visualWaypoints = {}
	for _, waypoint in ipairs(waypoints) do
		local visualWaypointClone = visualWaypoint:Clone()
		visualWaypointClone.Position = waypoint.Position
		visualWaypointClone.Parent = workspace
		visualWaypointClone.Color =
			(waypoint == waypoints[#waypoints] and Color3.fromRGB(0, 255, 0))
			or (waypoint.Action == Enum.PathWaypointAction.Jump and Color3.fromRGB(255, 0, 0))
			or Color3.fromRGB(255, 139, 0)
		table.insert(visualWaypoints, visualWaypointClone)
	end
	return visualWaypoints
end

--Destroy visual waypoints
local function destroyVisualWaypoints(waypoints)
	if waypoints then
		for _, waypoint in ipairs(waypoints) do
			waypoint:Destroy()
		end
	end
	return
end

--Get initial waypoint for non-humanoid
local function getNonHumanoidWaypoint(self)
	--Account for multiple waypoints that are sometimes in the same place
	for i = 2, #self._waypoints do
		if (self._waypoints[i].Position - self._waypoints[i - 1].Position).Magnitude > 0.1 then
			return i
		end
	end
	return 2
end

--Make NPC jump
local function setJumpState(self)
	pcall(function()
		if self._humanoid:GetState() ~= Enum.HumanoidStateType.Jumping and self._humanoid:GetState() ~= Enum.HumanoidStateType.Freefall then
			self._humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
		end
	end)
end

--Primary move function
local function move(self)
	if self._waypoints[self._currentWaypoint].Action == Enum.PathWaypointAction.Jump then
		setJumpState(self)
	end
	self._humanoid:MoveTo(self._waypoints[self._currentWaypoint].Position)
end

--Disconnect MoveToFinished connection when pathfinding ends
local function disconnectMoveConnection(self)
	self._moveConnection:Disconnect()
	self._moveConnection = nil
end

--Fire the WaypointReached event
local function invokeWaypointReached(self)
	local lastWaypoint = self._waypoints[self._currentWaypoint - 1]
	local nextWaypoint = self._waypoints[self._currentWaypoint]
	self._events.WaypointReached:Fire(self._agent, lastWaypoint, nextWaypoint)
end

local function moveToFinished(self, reached)
	
	--Stop execution if Path is destroyed
	if not getmetatable(self) then return end

	--Handle case for non-humanoids
	if not self._humanoid then
		if reached and self._currentWaypoint + 1 <= #self._waypoints then
			invokeWaypointReached(self)
			self._currentWaypoint += 1
		elseif reached then
			self._visualWaypoints = destroyVisualWaypoints(self._visualWaypoints)
			self._target = nil
			self._events.Reached:Fire(self._agent, self._waypoints[self._currentWaypoint])
		else
			self._visualWaypoints = destroyVisualWaypoints(self._visualWaypoints)
			self._target = nil
			declareError(self, self.ErrorType.TargetUnreachable)
		end
		return
	end

	if reached and self._currentWaypoint + 1 <= #self._waypoints  then --Waypoint reached
		if self._currentWaypoint + 1 < #self._waypoints then
			invokeWaypointReached(self)
		end
		self._currentWaypoint += 1
		move(self)
	elseif reached then --Target reached, pathfinding ends
		disconnectMoveConnection(self)
		self._status = Path.StatusType.Idle
		self._visualWaypoints = destroyVisualWaypoints(self._visualWaypoints)
		self._events.Reached:Fire(self._agent, self._waypoints[self._currentWaypoint])
	else --Target unreachable
		disconnectMoveConnection(self)
		self._status = Path.StatusType.Idle
		self._visualWaypoints = destroyVisualWaypoints(self._visualWaypoints)
		declareError(self, self.ErrorType.TargetUnreachable)
	end
end

--Refer to Settings.COMPARISON_CHECKS
local function comparePosition(self)
	if self._currentWaypoint == #self._waypoints then return end
	self._position._count = ((self._agent.PrimaryPart.Position - self._position._last).Magnitude <= 0.07 and (self._position._count + 1)) or 0
	self._position._last = self._agent.PrimaryPart.Position
	if self._position._count >= self._settings.COMPARISON_CHECKS then
		if self._settings.JUMP_WHEN_STUCK then
			setJumpState(self)
		end
		declareError(self, self.ErrorType.AgentStuck)
	end
end

--[[ STATIC METHODS ]]--
function Path.GetNearestCharacter(fromPosition)
	local character, dist = nil, math.huge
	for _, player in ipairs(Players:GetPlayers()) do
		if player.Character and (player.Character.PrimaryPart.Position - fromPosition).Magnitude < dist then
			character, dist = player.Character, (player.Character.PrimaryPart.Position - fromPosition).Magnitude
		end
	end
	return character
end

--[[ CONSTRUCTOR ]]--
function Path.new(agent, agentParameters, override)
	if not (agent and agent:IsA("Model") and agent.PrimaryPart) then
		output(error, "Pathfinding agent must be a valid Model Instance with a set PrimaryPart.")
	end

	local self = setmetatable({
		_settings = override or DEFAULT_SETTINGS;
		_events = {
			Reached = Instance.new("BindableEvent");
			WaypointReached = Instance.new("BindableEvent");
			Blocked = Instance.new("BindableEvent");
			Error = Instance.new("BindableEvent");
			Stopped = Instance.new("BindableEvent");
		};
		_agent = agent;
		_humanoid = agent:FindFirstChildOfClass("Humanoid");
		_path = PathfindingService:CreatePath(agentParameters);
		_status = "Idle";
		_t = 0;
		_position = {
			_last = Vector3.new();
			_count = 0;
		};
	}, Path)

	--Configure settings
	for setting, value in pairs(DEFAULT_SETTINGS) do
		self._settings[setting] = self._settings[setting] == nil and value or self._settings[setting]
	end

	--Path blocked connection
	self._path.Blocked:Connect(function(...)
		if (self._currentWaypoint <= ... and self._currentWaypoint + 1 >= ...) and self._humanoid then
			setJumpState(self)
			self._events.Blocked:Fire(self._agent, self._waypoints[...])
		end
	end)

	return self
end


--[[ NON-STATIC METHODS ]]--
function Path:Destroy()
	for _, event in ipairs(self._events) do
		event:Destroy()
	end
	self._events = nil
	if rawget(self, "_visualWaypoints") then
		self._visualWaypoints = destroyVisualWaypoints(self._visualWaypoints)
	end
	self._path:Destroy()
	setmetatable(self, nil)
	for k, _ in pairs(self) do
		self[k] = nil
	end
end

function Path:Stop()
	if not self._humanoid then
		output(error, "Attempt to call Path:Stop() on a non-humanoid.")
		return
	end
	if self._status == Path.StatusType.Idle then
		output(function(m)
			warn(debug.traceback(m))
		end, "Attempt to run Path:Stop() in idle state")
		return
	end
	disconnectMoveConnection(self)
	self._status = Path.StatusType.Idle
	self._visualWaypoints = destroyVisualWaypoints(self._visualWaypoints)
	self._events.Stopped:Fire(self._model)
end

function Path:Run(target)

	--Non-humanoid handle case
	if not target and not self._humanoid and self._target then
		moveToFinished(self, true)
		return
	end

	--Parameter check
	if not (target and (typeof(target) == "Vector3" or target:IsA("BasePart"))) then
		output(error, "Pathfinding target must be a valid Vector3 or BasePart.")
	end

	--Refer to Settings.TIME_VARIANCE
	if os.clock() - self._t <= self._settings.TIME_VARIANCE and self._humanoid then
		task.wait(os.clock() - self._t)
		declareError(self, self.ErrorType.LimitReached)
		return false
	elseif self._humanoid then
		self._t = os.clock()
	end

	--Compute path
	local pathComputed, _ = pcall(function()
		self._path:ComputeAsync(self._agent.PrimaryPart.Position, (typeof(target) == "Vector3" and target) or target.Position)
	end)

	--Make sure path computation is successful
	if not pathComputed
		or self._path.Status == Enum.PathStatus.NoPath
		or #self._path:GetWaypoints() < 2
		or (self._humanoid and self._humanoid:GetState() == Enum.HumanoidStateType.Freefall) then
		self._visualWaypoints = destroyVisualWaypoints(self._visualWaypoints)
		task.wait()
		declareError(self, self.ErrorType.ComputationError)
		return false
	end

	--Set status to active; pathfinding starts
	self._status = (self._humanoid and Path.StatusType.Active) or Path.StatusType.Idle
	self._target = target

	--Set network owner to server to prevent "hops"
	pcall(function()
		self._agent.PrimaryPart:SetNetworkOwner(nil)
	end)

	--Initialize waypoints
	self._waypoints = self._path:GetWaypoints()
	self._currentWaypoint = 2

	--Refer to Settings.COMPARISON_CHECKS
	if self._humanoid then
		comparePosition(self)
	end

	--Visualize waypoints
	destroyVisualWaypoints(self._visualWaypoints)
	self._visualWaypoints = (self.Visualize and createVisualWaypoints(self._waypoints))

	--Create a new move connection if it doesn't exist already
	self._moveConnection = self._humanoid and (self._moveConnection or self._humanoid.MoveToFinished:Connect(function(...)
		moveToFinished(self, ...)
	end))

	--Begin pathfinding
	if self._humanoid then
		self._humanoid:MoveTo(self._waypoints[self._currentWaypoint].Position)
